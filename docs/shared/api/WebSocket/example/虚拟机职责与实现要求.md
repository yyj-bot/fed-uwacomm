# 虚拟机职责与实现要求

本文档详细描述了虚拟机在联邦学习过程中的核心职责和具体实现要求，强化虚拟机作为纯执行者的"手脚"角色。

## 1. 核心职责概述

虚拟机作为联邦学习的执行单元，严格按照后端指令执行训练任务，不做任何主动决策：

### 1.1 主要职责
- **指令响应**: 接收并执行后端发送的所有WebSocket指令
- **本地训练**: 根据指令参数执行本地模型训练
- **梯度上传**: 训练完成后上传梯度数据到后端
- **状态报告**: 通过心跳和进度消息报告自身状态
- **多任务支持**: 根据taskId隔离并同时执行多个训练任务

### 1.2 绝对不负责的内容
- **任务编排决策**: 完全由后端控制
- **轮次同步决策**: 完全由后端控制
- **模型聚合逻辑**: 完全由后端执行
- **全局状态管理**: 完全由后端维护
- **其他虚拟机协调**: 禁止虚拟机间直接通信
- **训练策略调整**: 完全按后端配置执行

### 1.3 设计理念
- **完全被动**: 只响应后端指令，从不主动发起任务操作
- **精确执行**: 严格按照taskId隔离，精确执行指定任务
- **状态透明**: 实时上报状态，让后端掌握完整情况
- **无状态转换**: 不维护复杂状态机，状态由后端管理

```python
class FederatedLearningClient:
    def __init__(self, server_url: str, access_token: str, vm_id: str):
        self.server_url = server_url
        self.access_token = access_token
        self.vm_id = vm_id
        self.websocket = None
        self.is_connected = False
        self.active_tasks = {}  # taskId -> task_context

    def connect(self):
        """建立WebSocket连接"""
        try:
            url = f"{self.server_url}?token={self.access_token}&vmId={self.vm_id}"

            self.websocket = websocket.WebSocketApp(
                url,
                on_open=self._on_open,
                on_message=self._on_message,
                on_error=self._on_error,
                on_close=self._on_close
            )

            self.websocket.run_forever()

        except Exception as e:
            logger.error(f"连接服务器失败: {e}")

    def _on_open(self, ws):
        """连接建立回调"""
        logger.info("WebSocket连接已建立")
        self.is_connected = True

        # 发送连接确认消息
        connect_message = {
            "type": "CONNECT",
            "id": self._generate_message_id(),
            "timestamp": self._get_current_timestamp(),
            "vmId": self.vm_id,
            "data": {
                "vmId": self.vm_id,
                "capabilities": self._get_vm_capabilities()
            }
        }

        self._send_message(connect_message)

        # 启动心跳机制
        self._start_heartbeat()

    def _get_vm_capabilities(self) -> Dict[str, Any]:
        """获取虚拟机能力信息"""
        import platform
        import psutil

        return {
            "systemInfo": {
                "os": platform.system() + " " + platform.release(),
                "python": platform.python_version(),
                "memory": f"{psutil.virtual_memory().total // (1024**3)}GB",
                "cpu_cores": psutil.cpu_count()
            },
            "maxConcurrentTasks": 3,  # 最大并发任务数
            "supportedAlgorithms": [
                "FEDERATED_AVERAGING",
                "FEDERATED_PROXIMAL",
                "FEDERATED_NOVA",
                "SCAFFOLD"
            ]
        }
```

#### 2.1.2 心跳机制
**职责**:
- 定期发送心跳消息维持连接
- 报告当前资源使用状态
- 报告所有活跃任务状态

**实现要求**:
```python
def _start_heartbeat(self):
    """启动心跳机制"""
    def heartbeat_loop():
        while self.is_connected:
            try:
                heartbeat_message = {
                    "type": "HEARTBEAT",
                    "id": self._generate_message_id(),
                    "timestamp": self._get_current_timestamp(),
                    "vmId": self.vm_id,
                    "data": {
                        "status": "ACTIVE",
                        "resourceUsage": self._get_resource_usage(),
                        "activeTasks": self._get_active_tasks_status()
                    }
                }

                self._send_message(heartbeat_message)
                time.sleep(30)  # 30秒间隔

            except Exception as e:
                logger.error(f"心跳发送失败: {e}")

    threading.Thread(target=heartbeat_loop, daemon=True).start()

def _get_active_tasks_status(self) -> List[Dict[str, Any]]:
    """获取所有活跃任务状态"""
    task_status = []
    for task_id, task_context in self.active_tasks.items():
        task_status.append({
            "taskId": task_id,
            "status": task_context.get("status", "UNKNOWN"),
            "currentRound": task_context.get("current_round", 0),
            "progress": task_context.get("progress", 0.0),
            "lastActivity": task_context.get("last_activity", 0)
        })
    return task_status
```

```python
def _on_message(self, ws, message):
    """处理接收的消息"""
    try:
        msg = json.loads(message)
        msg_type = msg.get("type")

        # 根据消息类型处理
        if msg_type == "FEDERATED_TASK_START":
            self._handle_task_start(msg)
        elif msg_type == "ROUND_START":
            self._handle_round_start(msg)
        elif msg_type == "GLOBAL_MODEL_BROADCAST":
            self._handle_global_model_broadcast(msg)
        elif msg_type == "ROUND_COMPLETE":
            self._handle_round_complete(msg)
        elif msg_type == "FEDERATED_TASK_STOP":
            self._handle_task_stop(msg)
        elif msg_type == "CONNECT_ACK":
            self._handle_connect_ack(msg)
        elif msg_type == "HEARTBEAT_ACK":
            self._handle_heartbeat_ack(msg)
        else:
            logger.warning(f"未知消息类型: {msg_type}")

    except Exception as e:
        logger.error(f"消息处理失败: {e}")

def _handle_task_start(self, message: Dict[str, Any]):
    """处理任务启动指令"""
    try:
        task_data = message["data"]
        task_id = task_data["taskId"]

        # 初始化任务上下文
        self.active_tasks[task_id] = {
            "task_id": task_id,
            "status": "INITIALIZING",
            "config": task_data,
            "current_round": 0,
            "progress": 0.0,
            "last_activity": time.time(),
            "trainer": None
        }

        # 初始化训练器
        success = self._initialize_task_trainer(task_id, task_data)

        # 发送确认消息
        ack_message = {
            "type": "FEDERATED_TASK_START_ACK",
            "id": self._generate_message_id(),
            "timestamp": self._get_current_timestamp(),
            "vmId": self.vm_id,
            "data": {
                "taskId": task_id,
                "status": "SUCCESS" if success else "FAILED",
                "message": "任务初始化完成" if success else "任务初始化失败"
            }
        }

        self._send_message(ack_message)

        if success:
            self.active_tasks[task_id]["status"] = "READY"
        else:
            del self.active_tasks[task_id]

    except Exception as e:
        logger.error(f"处理任务启动失败: {e}")
```

```python
def _initialize_task_trainer(self, task_id: str, task_config: Dict[str, Any]) -> bool:
    """为指定任务初始化训练器"""
    try:
        # 提取配置参数
        algorithm = task_config.get("federatedAlgorithm")
        total_rounds = task_config.get("totalRounds")
        initial_model = task_config.get("initialGlobalModel")
        training_config = task_config.get("localTrainingConfig", {})

        # 创建任务专用的训练器
        trainer = TaskTrainer(
            task_id=task_id,
            algorithm=algorithm,
            config=training_config
        )

        # 加载初始全局模型
        if initial_model:
            trainer.load_global_model(initial_model)

        # 加载本地训练数据
        dataset_id = training_config.get("datasetId")
        if dataset_id:
            trainer.load_training_data(dataset_id)

        # 保存训练器到任务上下文
        self.active_tasks[task_id]["trainer"] = trainer

        logger.info(f"任务 {task_id} 训练器初始化成功")
        return True

    except Exception as e:
        logger.error(f"任务 {task_id} 训练器初始化失败: {e}")
        return False

class TaskTrainer:
    """任务专用训练器"""
    def __init__(self, task_id: str, algorithm: str, config: Dict[str, Any]):
        self.task_id = task_id
        self.algorithm = algorithm
        self.config = config
        self.model = None
        self.training_data = None
        self.is_training = False

    def load_global_model(self, model_data: Dict[str, Any]):
        """加载全局模型"""
        # 根据算法类型创建模型
        if self.algorithm == "FEDERATED_AVERAGING":
            self.model = self._create_fedavg_model(model_data)
        elif self.algorithm == "FEDERATED_PROXIMAL":
            self.model = self._create_fedprox_model(model_data)
        # ... 其他算法

    def load_training_data(self, dataset_id: str):
        """加载训练数据"""
        # 根据dataset_id加载对应的本地数据
        self.training_data = self._load_dataset(dataset_id)
```

**职责**:
- 接收ROUND_START指令开始训练
- 执行指定轮次的本地训练
- 训练完成后自动上传梯度

**实现要求**:
```python
def _handle_round_start(self, message: Dict[str, Any]):
    """处理轮次开始指令"""
    try:
        round_data = message["data"]
        task_id = round_data["taskId"]
        round_number = round_data["roundNumber"]

        # 验证任务存在
        if task_id not in self.active_tasks:
            logger.error(f"任务 {task_id} 不存在")
            return

        task_context = self.active_tasks[task_id]
        trainer = task_context["trainer"]

        # 发送轮次开始确认
        ack_message = {
            "type": "ROUND_START_ACK",
            "id": self._generate_message_id(),
            "timestamp": self._get_current_timestamp(),
            "vmId": self.vm_id,
            "data": {
                "taskId": task_id,
                "roundNumber": round_number,
                "status": "SUCCESS",
                "message": "轮次开始确认"
            }
        }
        self._send_message(ack_message)

        # 更新任务状态
        task_context["current_round"] = round_number
        task_context["status"] = "TRAINING"
        task_context["last_activity"] = time.time()

        # 启动本地训练 (异步执行)
        threading.Thread(
            target=self._execute_local_training,
            args=(task_id, round_number, round_data),
            daemon=True
        ).start()

    except Exception as e:
        logger.error(f"处理轮次开始失败: {e}")

def _execute_local_training(self, task_id: str, round_number: int, round_config: Dict[str, Any]):
    """执行本地训练"""
    try:
        task_context = self.active_tasks[task_id]
        trainer = task_context["trainer"]

        # 开始训练
        trainer.is_training = True
        training_result = trainer.train_round(round_config)

        # 训练完成，提取梯度
        gradients = trainer.extract_gradients()

        # 更新任务状态
        task_context["status"] = "GRADIENT_READY"
        task_context["last_activity"] = time.time()

        # 自动上传梯度
        self._upload_gradients(task_id, round_number, gradients, training_result)

    except Exception as e:
        logger.error(f"任务 {task_id} 轮次 {round_number} 训练失败: {e}")
        self._report_training_error(task_id, round_number, str(e))

def _upload_gradients(self, task_id: str, round_number: int, gradients: Dict, training_result: Dict):
    """上传梯度数据"""
    try:
        # 序列化梯度数据
        gradient_bytes = self._serialize_gradients(gradients)

        # 构建上传消息
        upload_message = {
            "type": "GRADIENT_UPLOAD",
            "id": self._generate_message_id(),
            "timestamp": self._get_current_timestamp(),
            "vmId": self.vm_id,
            "data": {
                "taskId": task_id,
                "roundNumber": round_number,
                "vmId": self.vm_id,
                "gradientData": base64.b64encode(gradient_bytes).decode('utf-8'),
                "trainingMetrics": {
                    "loss": training_result.get("final_loss", 0.0),
                    "accuracy": training_result.get("final_accuracy", 0.0),
                    "samplesCount": training_result.get("samples_count", 0),
                    "trainingTime": training_result.get("training_time", 0.0)
                }
            }
        }

        self._send_message(upload_message)

        # 更新任务状态为等待
        self.active_tasks[task_id]["status"] = "WAITING"

        logger.info(f"任务 {task_id} 轮次 {round_number} 梯度上传完成")

    except Exception as e:
        logger.error(f"梯度上传失败: {e}")
```

```python
def _handle_global_model_broadcast(self, message: Dict[str, Any]):
    """处理全局模型广播"""
    try:
        model_data = message["data"]
        task_id = model_data["taskId"]
        round_number = model_data["roundNumber"]

        # 验证任务存在
        if task_id not in self.active_tasks:
            logger.error(f"任务 {task_id} 不存在")
            return

        task_context = self.active_tasks[task_id]
        trainer = task_context["trainer"]

        # 解码全局模型数据
        global_model_bytes = base64.b64decode(model_data["globalModel"])
        global_model = self._deserialize_model(global_model_bytes)

        # 更新本地模型
        success = trainer.update_global_model(global_model)

        # 发送确认消息
        ack_message = {
            "type": "GLOBAL_MODEL_BROADCAST_ACK",
            "id": self._generate_message_id(),
            "timestamp": self._get_current_timestamp(),
            "vmId": self.vm_id,
            "data": {
                "taskId": task_id,
                "roundNumber": round_number,
                "status": "SUCCESS" if success else "FAILED",
                "message": "全局模型更新成功" if success else "全局模型更新失败"
            }
        }

        self._send_message(ack_message)

        if success:
            # 更新任务状态为就绪
            task_context["status"] = "READY"
            task_context["last_activity"] = time.time()

        logger.info(f"任务 {task_id} 全局模型更新完成")

    except Exception as e:
        logger.error(f"处理全局模型广播失败: {e}")
```

```python
class MultiTaskManager:
    """多任务管理器"""
    def __init__(self, max_concurrent_tasks: int = 3):
        self.max_concurrent_tasks = max_concurrent_tasks
        self.active_tasks = {}  # taskId -> TaskContext

    def can_accept_new_task(self) -> bool:
        """检查是否可以接受新任务"""
        return len(self.active_tasks) < self.max_concurrent_tasks

    def add_task(self, task_id: str, task_config: Dict[str, Any]) -> bool:
        """添加新任务"""
        if not self.can_accept_new_task():
            return False

        self.active_tasks[task_id] = TaskContext(task_id, task_config)
        return True

    def remove_task(self, task_id: str):
        """移除任务"""
        if task_id in self.active_tasks:
            task_context = self.active_tasks[task_id]
            task_context.cleanup()
            del self.active_tasks[task_id]

    def get_task_status(self) -> List[Dict[str, Any]]:
        """获取所有任务状态"""
        return [
            {
                "taskId": task_id,
                "status": task_context.status,
                "currentRound": task_context.current_round,
                "progress": task_context.progress
            }
            for task_id, task_context in self.active_tasks.items()
        ]

class TaskContext:
    """任务上下文"""
    def __init__(self, task_id: str, config: Dict[str, Any]):
        self.task_id = task_id
        self.config = config
        self.status = "INITIALIZING"
        self.current_round = 0
        self.progress = 0.0
        self.trainer = None
        self.last_activity = time.time()

    def cleanup(self):
        """清理任务资源"""
        if self.trainer:
            self.trainer.cleanup()
```

```python
def _report_training_error(self, task_id: str, round_number: int, error_message: str):
    """报告训练错误"""
    error_msg = {
        "type": "ERROR",
        "id": self._generate_message_id(),
        "timestamp": self._get_current_timestamp(),
        "vmId": self.vm_id,
        "data": {
            "taskId": task_id,
            "roundNumber": round_number,
            "errorType": "TRAINING_ERROR",
            "errorMessage": error_message,
            "severity": "ERROR",
            "context": {
                "vmResources": self._get_resource_usage(),
                "taskStatus": self.active_tasks.get(task_id, {}).get("status", "UNKNOWN")
            }
        }
    }

    self._send_message(error_msg)

    # 更新任务状态为错误
    if task_id in self.active_tasks:
        self.active_tasks[task_id]["status"] = "ERROR"
        self.active_tasks[task_id]["last_activity"] = time.time()

def _handle_connection_lost(self):
    """处理连接丢失"""
    logger.warning("WebSocket连接丢失，尝试重连...")

    self.is_connected = False

    # 暂停所有训练任务
    for task_id, task_context in self.active_tasks.items():
        if task_context.get("trainer"):
            task_context["trainer"].pause_training()

    # 尝试重连
    self._reconnect()

def _reconnect(self):
    """重连机制"""
    max_retries = 5
    retry_delay = 1

    for attempt in range(max_retries):
        try:
            time.sleep(retry_delay)
            logger.info(f"重连尝试 {attempt + 1}/{max_retries}")

            self.connect()

            if self.is_connected:
                logger.info("重连成功")
                # 恢复所有任务的训练
                self._resume_all_tasks()
                return True

            retry_delay = min(retry_delay * 2, 60)  # 指数退避

        except Exception as e:
            logger.error(f"重连失败: {e}")

    logger.error("所有重连尝试失败")
    return False
```

## 3. 关键实现特点

### 3.1 完全被动响应
- 虚拟机从不主动发起任务操作
- 所有操作都是对后端指令的响应
- 通过taskId实现精确的任务级控制

### 3.2 状态透明化
- 通过心跳消息实时报告所有任务状态
- 训练进度实时可见
- 异常情况立即上报

### 3.3 资源隔离
- 每个taskId对应独立的训练上下文
- 避免不同任务间的资源冲突
- 支持同时执行多个训练任务

### 3.4 简化的错误处理
- 只报告错误，不做恢复决策
- 所有恢复策略由后端控制
- 维持连接稳定性是首要任务

## 4. 性能要求

### 4.1 响应性能
- WebSocket消息响应: < 100ms
- 任务启动时间: < 5秒
- 模型更新时间: < 3秒

### 4.2 训练性能
- 支持最多3个并发训练任务
- 内存使用不超过系统可用内存的80%
- 训练完成后立即上传梯度

### 4.3 稳定性
- 连续运行时间: 支持7x24小时
- 自动重连: 网络中断后自动恢复
- 状态一致性: 与后端状态完全同步

这个实现确保了虚拟机作为联邦学习系统的"手脚"，完全按照后端"大脑"的指令执行，提供可靠的训练执行能力。
