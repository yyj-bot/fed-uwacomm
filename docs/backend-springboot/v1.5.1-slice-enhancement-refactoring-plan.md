**示例**:
```java
@Service
public class IIDDataSlicingServiceImpl implements DataSlicingService {
    @Override
    public List<DataSlice> sliceDataset(String datasetId, List<String> vmIds, AllocationStrategy strategy) {
        // 1. 读取原始数据集大小
        int totalSamples = getDatasetSize(datasetId);
        int vmCount = vmIds.size();

        // 2. 计算每个VM的切片大小
        int baseSize = totalSamples / vmCount;
        int remainder = totalSamples % vmCount;

        // 3. 为每个VM生成切片
        List<DataSlice> slices = new ArrayList<>();
        int currentIndex = 0;

        for (int i = 0; i < vmCount; i++) {
            int sliceSize = baseSize + (i < remainder ? 1 : 0);
            int startIndex = currentIndex;
            int endIndex = currentIndex + sliceSize - 1;

            SliceInfo sliceInfo = SliceInfo.builder()
                .startIndex(startIndex)
                .endIndex(endIndex)
                .sliceSamples(sliceSize)
                .totalSamples(totalSamples)
                .sliceIndex(i + 1)
                .totalSlices(vmCount)
                .allocationStrategy("IID")
                .build();

            slices.add(new DataSlice(vmIds.get(i), sliceInfo));
            currentIndex += sliceSize;
        }

        return slices;
    }
}
```

  ```java
          .localEndIndex(localEndIndex)
          .globalStartIndex(sliceInfo.getStartIndex() + localStartIndex)
          .globalEndIndex(sliceInfo.getStartIndex() + localEndIndex)
  ```

  ```java
  public ContinuityCheck checkContinuity(List<Integer> receivedIndices) {
      Collections.sort(receivedIndices);
      List<GapRange> gaps = new ArrayList<>();

      for (int i = 1; i < receivedIndices.size(); i++) {
          int prev = receivedIndices.get(i - 1);
          int curr = receivedIndices.get(i);

          if (curr - prev > 1) {
              gaps.add(new GapRange(prev + 1, curr - 1));
          }
      }

      return ContinuityCheck.builder()
          .hasGaps(!gaps.isEmpty())
          .gapRanges(gaps)
          .build();
  }
  ```

  ```java
  public VerificationDecision decideOnVerificationFailure(String vmId, SliceVerification verification) {
      if (verification.getIsComplete()) {
          return VerificationDecision.ACCEPT;
      }

      int missingCount = verification.getMissingIndices().size();
      int totalSamples = verification.getExpectedSamples();
      double missingRate = (double) missingCount / totalSamples;

      if (missingRate < 0.05) {  // 缺失率 < 5%
          return VerificationDecision.RETRY_MISSING;
      } else if (missingRate < 0.20) {  // 缺失率 < 20%
          return VerificationDecision.EXCLUDE_VM;
      } else {
          return VerificationDecision.ABORT_TASK;
      }
  }
  ```

  ```java
  public SliceVerification generateSliceVerification() {
      List<Integer> receivedIndices = new ArrayList<>(receivedDataMap.keySet());

      // 计算实际接收范围
      int actualStartIndex = receivedIndices.stream().min(Integer::compare).orElse(-1);
      int actualEndIndex = receivedIndices.stream().max(Integer::compare).orElse(-1);
      int actualSamples = receivedIndices.size();

      // 检测缺失索引
      List<Integer> missingIndices = new ArrayList<>();
      for (int i = receivedSliceInfo.getStartIndex(); i <= receivedSliceInfo.getEndIndex(); i++) {
          if (!receivedDataMap.containsKey(i)) {
              missingIndices.add(i);
          }
      }

      // 检测间隙
      ContinuityCheck continuityCheck = checkContinuity(receivedIndices);

      // 判断是否完整
      boolean isComplete = actualSamples == receivedSliceInfo.getSliceSamples() &&
                          missingIndices.isEmpty() &&
                          !continuityCheck.getHasGaps();

      return SliceVerification.builder()
          .expectedStartIndex(receivedSliceInfo.getStartIndex())
          .expectedEndIndex(receivedSliceInfo.getEndIndex())
          .expectedSamples(receivedSliceInfo.getSliceSamples())
          .actualStartIndex(actualStartIndex)
          .actualEndIndex(actualEndIndex)
          .actualSamples(actualSamples)
          .isComplete(isComplete)
          .missingIndices(missingIndices)
          .continuityCheck(continuityCheck)
          .build();
  }
  ```
